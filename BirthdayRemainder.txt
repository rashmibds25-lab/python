import json
from datetime import datetime, date, timedelta

FILE = "birthdays.json"


def load_birthdays():
    try:
        with open(FILE, "r") as f:
            return json.load(f)
    except FileNotFoundError:
        return []


def save_birthdays(data):
    with open(FILE, "w") as f:
        json.dump(data, f, indent=4)


def add_birthday(data):
    name = input("Enter name: ")
    birthday = input("Enter birthday (DD-MM-YYYY): ")
    relation = input("Enter relation: ")

    try:
        datetime.strptime(birthday, "%d-%m-%Y")
        data.append({
            "name": name,
            "birthday": birthday,
            "relation": relation
        })
        save_birthdays(data)
        print("Birthday added successfully!")
    except ValueError:
        print("Invalid date format!")


def search_birthday(data):
    name = input("Enter name to search: ")

    for person in data:
        if person["name"].lower() == name.lower():
            print("Name:", person["name"])
            print("Birthday:", person["birthday"])
            print("Relation:", person["relation"])
            return

    print("Birthday not found!")


def update_birthday(data):
    name = input("Enter name to update: ")

    for person in data:
        if person["name"].lower() == name.lower():
            person["birthday"] = input("Enter new birthday (DD-MM-YYYY): ")
            person["relation"] = input("Enter new relation: ")
            save_birthdays(data)
            print("Birthday updated successfully!")
            return

    print("Birthday not found!")


def delete_birthday(data):
    name = input("Enter name to delete: ")

    for person in data:
        if person["name"].lower() == name.lower():
            data.remove(person)
            save_birthdays(data)
            print("Birthday deleted successfully!")
            return

    print("Birthday not found!")


def upcoming_birthdays(data):
    today = date.today()
    end_date = today + timedelta(days=7)

    print("\nUpcoming Birthdays:")

    found = False

    for person in data:
        birthday = datetime.strptime(
            person["birthday"], "%d-%m-%Y"
        ).date()

        next_birthday = birthday.replace(year=today.year)

        if next_birthday < today:
            next_birthday = next_birthday.replace(
                year=today.year + 1
            )

        if today <= next_birthday <= end_date:
            print(person["name"], "-", person["birthday"])
            found = True

    if not found:
        print("No upcoming birthdays!")


def main():
    data = load_birthdays()

    while True:
        print("\n===== BIRTHDAY REMINDER =====")
        print("1. Add Birthday")
        print("2. Update Birthday")
        print("3. Search Birthday")
        print("4. Delete Birthday")
        print("5. Show Upcoming Birthdays")
        print("6. Exit")

        choice = input("Enter your choice: ")

        if choice == "1":
            add_birthday(data)

        elif choice == "2":
            update_birthday(data)

        elif choice == "3":
            search_birthday(data)

        elif choice == "4":
            delete_birthday(data)

        elif choice == "5":
            upcoming_birthdays(data)

        elif choice == "6":
            print("Thank you!")
            break

        else:
            print("Invalid choice!")


main()
